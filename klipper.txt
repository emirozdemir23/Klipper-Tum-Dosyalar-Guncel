[gcode_macro PRINT_START]
gcode:
    # S1 SABIT-YATAK GUVENLIK KILIDI: Bu makro ARTIK G28 CAGIRMAZ.
    # Eski hali kor bir G28 atiyordu; G28, CALIBRATE_Z_OFFSET'in kurdugu
    # "gcode Z0 = temas yuzeyi" tabanini SILER ve dosyanin ilk mutlak hamlesi
    # (G0 Z0.2) KINEMATIK 0.2'ye gider — temas yuzeyi kinematik 0'in ustundeyse
    # igne yataga/petriye gomulur. Zorunlu operator sirasi:
    #     G28  ->  CALIBRATE_Z_OFFSET  ->  baski dosyasi.
    # Kosullar saglanmiyorsa action_raise_error dosyayi ILK HAREKETTEN ONCE iptal eder.
    {% if "xyz" not in printer.toolhead.homed_axes %}
        { action_raise_error("PRINT_START IPTAL: Eksenler home degil. Sira: G28 -> CALIBRATE_Z_OFFSET -> baski.") }
    {% endif %}
    {% if printer["gcode_macro Z_OFFSET_DURUM"].kalibre|int != 1 %}
        { action_raise_error("PRINT_START IPTAL: Z-offset kalibre edilmemis. CALIBRATE_Z_OFFSET calistir. (DIKKAT: G28 kalibrasyonu sifirlar - once G28, SONRA kalibrasyon.)") }
    {% endif %}
    G90                 ; mutlak konum modu
    RESPOND MSG="PRINT_START: home + Z-offset kalibrasyonu dogrulandi. Baski basliyor (SABIT YATAK modu)."
   

[gcode_macro PRINT_END]
#   Use PRINT_END for the slicer ending script - please customise for your slicer of choice
gcode:
    TURN_OFF_HEATERS               ; W1: EN ONCE calisir - asagidaki hareketlerden biri hata
                                   ; verip makroyu kesse bile isiticilar ACIK KALAMAZ
    M400                           ; hareket tamponunun bosalmasini bekle
    G92 E0                         ; extruder eksenini sifirla
    G1 E-0.5 F60                   ; W2: siringa olcekli retract (0.8mm/tur vidali mil).
                                   ; Eski E-10 F3600 = 4500 RPM -> garantili stall + 10mm piston kaybi idi.
    ; M107 KALDIRILDI: part-cooling [fan] yok (PA0/1/2 artik peltier fanlari).
    ; Peltier heatsink fanlari [heater_fan] ile OTOMATIK — komut gerekmez.
    {% if "xyz" in printer.toolhead.homed_axes %}
        # W1: Z-hop tavana kalan bosluga gore SINIRLANIR -> "Move out of range" iptali imkansiz.
        # Kinematik konum kullanilir (G92 ofsetinden etkilenmez); goreli +Z = yukari = guvenli.
        {% set z_max = printer.toolhead.axis_maximum.z | float %}
        {% set z_now = printer.toolhead.position.z | float %}
        {% set z_hop = [[z_max - z_now - 0.5, 10.0] | min, 0.0] | max %}
        {% if z_hop > 0.0 %}
            G91                    ; goreli mod (sadece Z-hop icin)
            G0 Z{z_hop} F600       ; 10mm/s (max_z_velocity 15 altinda)
        {% endif %}
        G90                        ; mutlak moda don
        G0 X125 Y115 F1500         ; park (limit ici: X<=230 Y<=120; W5: eski F20000 -> 25mm/s)
    {% else %}
        RESPOND TYPE=error MSG="PRINT_END: eksenler home degil - park/Z-hop atlandi (isiticilar zaten kapatildi)."
    {% endif %}

[gcode_macro KAFALARI_SIFIRLA]
description: Z1 ve Z2'yi sıfırlar ve konumlarını HAFIZAYA kaydeder.
gcode:
    # --- Z1 Sıfırla (soft-limit uyumlu homing) ---
    # SET_POSITION=-18 mevcut konumu -18 etiketler (SET_POSITION range-check'e girmez);
    # MOVE=82 => 82-(-18)=100mm fiziksel arama payi (eski davranisla AYNI) AMA MOVE
    # argumani 82 <= position_max(82) oldugu icin "Move out of range" YEMEZ. Endstop
    # tetikleyince SET_POSITION=82 ile tepe=82 capasi kurulur (logical 0 = fiziksel dip).
    # STOP_ON_ENDSTOP=home: eski =1 ile ayni davranis; deprecate uyarisini onler.
    MANUAL_STEPPER STEPPER=z1 SET_POSITION=-18
    MANUAL_STEPPER STEPPER=z1 MOVE=82 SPEED=3 STOP_ON_ENDSTOP=home
    MANUAL_STEPPER STEPPER=z1 SET_POSITION=82
    
    # --- Z2 Sıfırla (z1 ile ayni mantik; bkz. yukaridaki not) ---
    MANUAL_STEPPER STEPPER=z2 SET_POSITION=-18
    MANUAL_STEPPER STEPPER=z2 MOVE=82 SPEED=3 STOP_ON_ENDSTOP=home
    MANUAL_STEPPER STEPPER=z2 SET_POSITION=82

    SET_GCODE_VARIABLE MACRO=KAFA_HAFIZA VARIABLE=z1_konum VALUE=82.0
    SET_GCODE_VARIABLE MACRO=KAFA_HAFIZA VARIABLE=z2_konum VALUE=82.0

    MANUAL_STEPPER STEPPER=z1 MOVE=80 SPEED=3
    MANUAL_STEPPER STEPPER=z2 MOVE=80 SPEED=3

#           Primary homing rutini KAFALARI_SIFIRLA'dir.
# DIKKAT: Bu makro GANTRY'yi SABIT yataga dogru ASAGI surer. PROBE, [stepper_z]'yi
#         (PE5/PE4) PA13 loadcell tetiklenene kadar indirir (z1/z2 DEGIL).
# MIMARI (KULLANICI ONAYLI): Yatak SABIT. [stepper_z] (PE5/PE4) TUM GANTRY'yi yataga
#         dogru indirir/kaldirir. z1/z2 = BIREYSEL baski kafalari (ayri motorlar).
#         PROBE oncesi z1 ve z2 referans konumuna (80.0) getirilerek geometrik
#         tekrarlanabilirlik saglanir. Gantry PA13 "Ana Z" loadcell tetiklenene
#         kadar ASAGI surulur → gantry datumu (Z=0) kurulur.
#         position_min:0 = sert taban; motor Z=0 altina INEMEZ. Yine de ELIN
#         E-STOP'ta olsun: inis denetimli/supervised bir islemdir.
# =====================================================================================
[gcode_macro CALIBRATE_Z_OFFSET]
description: Yatak uzerinde prob ile Z=0 belirle. ELIN E-STOP'TA OLSUN!
gcode:
    {% if "xyz" not in printer.toolhead.homed_axes %}
        RESPOND MSG="HATA: Z-Offset icin once G28 ile home yapmalisin!"
    {% else %}
        RESPOND MSG="Kalibrasyon basladi: z1 ve z2 kafalari referans konumuna (80.0) getiriliyor..."

        # Kesin tekrarlanabilirlik icin bireysel kafalari bilinen referans noktasina cekiyoruz
        MANUAL_STEPPER STEPPER=z1 MOVE=80 SPEED=3
        MANUAL_STEPPER STEPPER=z2 MOVE=80 SPEED=3

        # Kafalarin konum durumunu hafizada guncelliyoruz
        SET_GCODE_VARIABLE MACRO=KAFA_HAFIZA VARIABLE=z1_konum VALUE=80.0
        SET_GCODE_VARIABLE MACRO=KAFA_HAFIZA VARIABLE=z2_konum VALUE=80.0

        RESPOND MSG="Kafalar referans konumunda. Yatak merkezine gidiliyor..."
        G90                  ; mutlak konum modu
        G1 X120 Y60 F1500    ; yatak merkezine git (X120 Y60 guvenli bolge; W5: 25mm/s)

        RESPOND MSG="Inise 3 saniye! Anormallik olursa E-STOP dugmesine bas."
        G4 P3000             ; 3 sn supervisor penceresi (bed-ward inis ONCESI)

        # PROBE [probe] blogunu kullanir; loadcell (PA13) tetiklenince durur.
        PROBE PROBE_SPEED=2 SAMPLES=1
        # ============ W6 GUVENLIK NOTU: G92 Z0 TABANI YENIDEN TANIMLAR ============
        # position_min:0 korumasi KINEMATIK uzaydadir. G92 Z0 sonrasi gcode Z0 =
        # temas yuzeyi olur; temas yuzeyi kinematik 0'in USTUNDEyse (or: petri kabi /
        # well-plate, yatak datumundan birkac mm yuksek), NEGATIF gcode Z komutlari
        # kinematik 0'a kadar KABUL EDILIR -> "G1 Z-2" gibi bir yazim hatasi kabi
        # ezebilir. Sert taban CIPLAK YATAK datumunu korur, problanan yuzeyi DEGIL.
        # Kalibrasyon sonrasi kurallar:
        #   1) ASLA negatif Z gonderme (GUI tarafi da negatif Z'yi reddetmeli),
        #   2) position_endstop:83 kalibrasyonunu guncel tut (kinematik 0 ~= ciplak yatak).
        G92 Z0               ; temas noktasi = Z0 (bu oturumluk, kalici degil)
        SET_GCODE_VARIABLE MACRO=Z_OFFSET_DURUM VARIABLE=kalibre VALUE=1   ; S1: PRINT_START kilidini ac
        G1 Z5 F300           ; yatagi/nozulu korumak icin hemen guvenli yukseklige kalk
        RESPOND MSG="Z-Sifiri belirlendi (bu oturum). DIKKAT: negatif Z, kinematik tabana kadar hala kabul edilir - problanan yuzeyin altina inme!"
    {% endif %}

[gcode_macro RECINELERI_SIFIRLA]
description: 3 recine asansorunu tavana ceker ve hafizayi 58'e ayarlar. (FORCE=1 ile guard asilir)
gcode:
    # --- GUVENLIK GUARD'I (over-travel onleme) + FORCE override ---
    # RECINE_HAFIZA degiskenlerinden biri >5.0 ise motorlar ZATEN yukarida demektir;
    # tekrar +58 surmek over-travel olur -> FORCE=0 (varsayilan) ise HIC hareket etmeden iptal.
    # FORCE=1 ile bilinckli asilabilir (risk operatorde). Once RECINE_MEMORY_RESET dusunulmeli.
    {% set force = params.FORCE | default(0) | int %}
    {% set r1 = printer["gcode_macro RECINE_HAFIZA"].recine1_konum | float %}
    {% set r2 = printer["gcode_macro RECINE_HAFIZA"].recine2_konum | float %}
    {% set r3 = printer["gcode_macro RECINE_HAFIZA"].recine3_konum | float %}
    {% if (r1 > 5.0 or r2 > 5.0 or r3 > 5.0) and force == 0 %}
        { action_respond_info("IPTAL: Recineler zaten yukarida (r1=%s, r2=%s, r3=%s). Hareket YOK. Asagi indir + RECINE_MEMORY_RESET, ya da bilerek RECINELERI_SIFIRLA FORCE=1." % (r1, r2, r3)) }
        M117 IPTAL: Recineler yukarida! Islem Iptal.
    {% else %}
        {% if force == 1 and (r1 > 5.0 or r2 > 5.0 or r3 > 5.0) %}
            RESPOND MSG="UYARI: FORCE=1 - guard bilinckli asildi (recineler yukarida olabilir). Over-travel riski operatorde!"
        {% endif %}
        RESPOND MSG="Recine asansorleri tavana cikiyor, operasyon basladi!"

        # 1. Tum recine motorlarini ana extruder motion queue'suna bagla
        SYNC_EXTRUDER_MOTION EXTRUDER=recine_1 MOTION_QUEUE=extruder
        SYNC_EXTRUDER_MOTION EXTRUDER=recine_2 MOTION_QUEUE=extruder
        SYNC_EXTRUDER_MOTION EXTRUDER=recine_3 MOTION_QUEUE=extruder

        # 2. Extruder eksenini sifirla
        G92 E0

        # 3. Motorlari senkronize sekilde hareket ettir.
        # KOMISYON-GUVENLI HIZ: F60 = 1 mm/s; rotation_distance 0.8 -> 75 motor RPM.
        # F300 (eski deger) yon, yuk, akim ve stall davranisi FIZIKSEL olarak
        # dogrulanana kadar GERI KONULMAMALI.
        # DIKKAT: Eger E58 komutu asansorleri "tavana" degil de "asagi" indiriyorsa,
        # yonleri ters cevirmek icin bu satiri 'G1 E-58 F60' olarak degistirmelisin.
        G1 E58 F60

        # 4. Extruder'i tekrar sifirla ve motorlari bosta birak (Guvenlik icin)
        # Sadece T0/T1/T2 secildiginde sirayla baglanmalilar.
        G92 E0
        SYNC_EXTRUDER_MOTION EXTRUDER=recine_1 MOTION_QUEUE=
        SYNC_EXTRUDER_MOTION EXTRUDER=recine_2 MOTION_QUEUE=
        SYNC_EXTRUDER_MOTION EXTRUDER=recine_3 MOTION_QUEUE=

        # 5. Hafiza degiskenlerini 58.0 olarak guncelle
        SET_GCODE_VARIABLE MACRO=RECINE_HAFIZA VARIABLE=recine1_konum VALUE=58.0
        SET_GCODE_VARIABLE MACRO=RECINE_HAFIZA VARIABLE=recine2_konum VALUE=58.0
        SET_GCODE_VARIABLE MACRO=RECINE_HAFIZA VARIABLE=recine3_konum VALUE=58.0

        RESPOND MSG="3 Ekstruder 58.0 konumuna ayarlandi ve kuyruktan cikarildi."
    {% endif %}

[gcode_macro RECINE_MEMORY_RESET]
description: Recine konum hafizalarini guvenli islem icin 0.0'a ceker.
gcode:
    # SADECE hafiza/bookkeeping sifirlamasi — MOTOR HAREKETI YOK.
    # Recine'lerin FIZIKSEL olarak asagida oldugunu DOGRULADIKTAN sonra calistir;
    # aksi halde RECINELERI_SIFIRLA bilinmeyen konumdan +58 surup over-travel yapabilir.
    SET_GCODE_VARIABLE MACRO=RECINE_HAFIZA VARIABLE=recine1_konum VALUE=0.0
    SET_GCODE_VARIABLE MACRO=RECINE_HAFIZA VARIABLE=recine2_konum VALUE=0.0
    SET_GCODE_VARIABLE MACRO=RECINE_HAFIZA VARIABLE=recine3_konum VALUE=0.0
    RESPOND MSG="Recine hafizasi 0.0 olarak sifirlandi. (Fiziksel olarak indirildigini DOGRULA!)"
    M117 Recine hafizasi sifirlandi (0.0)
