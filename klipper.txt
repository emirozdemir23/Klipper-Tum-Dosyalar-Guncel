[gcode_macro PRINT_START]

    RESPOND MSG="Basim islemi baslatiliyor (SABIT YATAK modu)..."
    G28                 ; SADECE homing. homing_override: Z tepeye(83)+KAFALARI_SIFIRLA+X/Y. Gantry yataktan UZAK.
    G90                 ; mutlak konum modu
    # WARNING: Manual Z-offset calibration required before absolute Z-moves.
    # SABIT YATAK: otomatik yaklasim/prob dizisi (G1 Z5, z1/z2 inisi, PROBE, G92 Z0, Z_GANTRY_HOMING)
    # GUVENLIK icin KALDIRILDI. Asagi/mutlak Z hareketinden ONCE Z-offset ELLE kalibre edilmeli.
    RESPOND MSG="Home tamam. UYARI: baski oncesi Z-offset elle kalibre edilmeli (otomatik prob devre disi)."
   

[gcode_macro PRINT_END]
#   Use PRINT_END for the slicer ending script - please customise for your slicer of choice
gcode:
    M400                           ; wait for buffer to clear
    G92 E0                         ; zero the extruder
    G1 E-10.0 F3600                ; retract filament
    G91                            ; relative positioning
    G0 Z1.00 X20.0 Y20.0 F20000    ; move nozzle to remove stringing
    TURN_OFF_HEATERS
    ; M107 KALDIRILDI: part-cooling [fan] yok (PA0/1/2 artik peltier fanlari).
    ; Peltier heatsink fanlari [heater_fan] ile OTOMATIK — komut gerekmez.
    G1 Z10 F500                     ; move nozzle up 2mm
    G90                            ; absolute positioning
    G0  X125 Y115 F500             ; park nozzle at rear (Y within 0-120)
